La bolsa, las opciones y Black-Scholes

Los fundamentos para después hacerlo en Python sabiendo qué hace cada línea: qué es una acción y su volatilidad, por qué la plata de hoy vale más que la de mañana, qué paga una opción, de dónde sale la fórmula de Black-Scholes y qué miden las griegas. Debajo de cada lección está el mismo cálculo en Python, con numpy y scipy.

Las empresas y los precios son inventados. Es material para aprender, no una recomendación de inversión.

1. Una acción y su precio

El mismo cálculo en Python 
import numpy as np

rng = np.random.default_rng(4)
S0, mu, sigma, dias = 100, 0.08, 0.20, 126
dt = 1 / 252                                    # un día de bolsa, en años

# El modelo de Black-Scholes: cada día el precio se multiplica por e^(algo al azar)
z = rng.standard_normal(dias)
precios = S0 * np.exp(np.cumsum((mu - sigma**2 / 2) * dt + sigma * np.sqrt(dt) * z))
precios = np.insert(precios, 0, S0)

simple = precios[1:] / precios[:-1] - 1        # (P1 - P0) / P0
logaritmico = np.log(precios[1:] / precios[:-1])  # ln(P1 / P0): se suman de un día al otro

volatilidad = logaritmico.std(ddof=1) * np.sqrt(252)   # desvío diario, pasado a un año
print(f"σ medida: {volatilidad:.1%}")

# Con datos de verdad (pandas): precios.pct_change() y np.log(precios).diff()

2. El valor del dinero en el tiempo

El mismo cálculo en Python 
import numpy as np

monto, r, T = 1000, 0.12, 5

anual = monto * (1 + r) ** T              # una vez por año
mensual = monto * (1 + r / 12) ** (12 * T)  # una vez por mes
continua = monto * np.exp(r * T)          # a cada instante: e^(rT)

promesa = 2000
hoy = promesa * np.exp(-r * T)            # valor presente: e^(-rT), el "descuento"
print(f"{anual:.2f} {mensual:.2f} {continua:.2f} -> hoy vale {hoy:.2f}")

4. De dónde sale Black-Scholes

El mismo cálculo en Python 
import numpy as np
from scipy.stats import norm

S, K, T, r, sigma = 100, 105, 0.5, 0.05, 0.25

# 1) Simulando: muchos futuros, el promedio de lo que paga, traído a hoy.
#    Se simula creciendo a la tasa sin riesgo r (el mundo "neutral al riesgo").
rng = np.random.default_rng(11)
z = rng.standard_normal(20_000)
ST = S * np.exp((r - sigma**2 / 2) * T + sigma * np.sqrt(T) * z)
precio_mc = np.exp(-r * T) * np.maximum(ST - K, 0).mean()

# 2) Con la fórmula
d1 = (np.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)
call = S * norm.cdf(d1) - K * np.exp(-r * T) * norm.cdf(d2)
put = K * np.exp(-r * T) * norm.cdf(-d2) - S * norm.cdf(-d1)

print(f"simulando: {precio_mc:.2f} · fórmula: {call:.2f} · put: {put:.2f}")
print(f"N(d2) = {norm.cdf(d2):.1%}: probabilidad (neutral al riesgo) de terminar con S > K")

# La volatilidad implícita: el σ que hace que la fórmula dé el precio de mercado
from scipy.optimize import brentq
def bs_call(s):
    a = (np.log(S / K) + (r + s**2 / 2) * T) / (s * np.sqrt(T))
    return S * norm.cdf(a) - K * np.exp(-r * T) * norm.cdf(a - s * np.sqrt(T))
precio_de_mercado = 8.00
print("σ implícita:", brentq(lambda s: bs_call(s) - precio_de_mercado, 0.01, 3))

5. Las griegas

El mismo cálculo en Python 
import numpy as np
from scipy.stats import norm

S, K, T, r, sigma = 100, 100, 0.5, 0.05, 0.25
d1 = (np.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * np.sqrt(T))
d2 = d1 - sigma * np.sqrt(T)

delta = norm.cdf(d1)                                  # por $1 de la acción
gamma = norm.pdf(d1) / (S * sigma * np.sqrt(T))       # cuánto cambia delta por $1
theta = (-S * norm.pdf(d1) * sigma / (2 * np.sqrt(T))
         - r * K * np.exp(-r * T) * norm.cdf(d2)) / 365   # por día
vega = S * norm.pdf(d1) * np.sqrt(T) / 100            # por punto (1 %) de σ
rho = K * T * np.exp(-r * T) * norm.cdf(d2) / 100     # por punto (1 %) de r

# Sin fórmula: diferencias finitas (mover un poquito y mirar cuánto cambia)
def call(S=S, T=T, r=r, sigma=sigma):
    a = (np.log(S / K) + (r + sigma**2 / 2) * T) / (sigma * np.sqrt(T))
    return S * norm.cdf(a) - K * np.exp(-r * T) * norm.cdf(a - sigma * np.sqrt(T))
h = 0.01
print(delta, (call(S + h) - call(S - h)) / (2 * h))   # dan lo mismo

6. Estrategias con opciones

El mismo cálculo en Python 
import numpy as np

S = np.linspace(65, 135, 701)                   # precios posibles al vencer

def call(K, prima, cantidad=1): return cantidad * (np.maximum(S - K, 0) - prima)
def put(K, prima, cantidad=1):  return cantidad * (np.maximum(K - S, 0) - prima)
def accion(precio_hoy, cantidad=1): return cantidad * (S - precio_hoy)

# Una estrategia es la suma de sus patas (las primas, de Black-Scholes)
put_protector = accion(100) + put(95, 2.35)
straddle = call(100, 5.60) + put(100, 4.36)
spread_alcista = call(100, 5.60) + call(115, 1.07, cantidad=-1)

empates = S[np.where(np.diff(np.sign(straddle)))[0]]
print("el straddle empata en", empates.round(1), "· pierde como mucho", straddle.min().round(2))
